4. C Program – Banker's Algorithm
Save the following program as bankers.c.

#include <stdio.h>

int main()
{
    int n = 5, m = 4;

    int alloc[5][4] = {
        {0, 0, 1, 2},
        {1, 0, 0, 0},
        {1, 3, 5, 4},
        {0, 6, 3, 2},
        {0, 0, 1, 4}
    };

    int max[5][4] = {
        {0, 0, 1, 2},
        {1, 7, 5, 0},
        {2, 3, 5, 6},
        {0, 6, 5, 2},
        {0, 6, 5, 6}
    };

    int avail[4] = {1, 5, 2, 0};

    int need[5][4];
    int finish[5] = {0};
    int safeSeq[5];

    int i, j, k = 0, count = 0;
    int found;

    // Calculate Need Matrix
    for(i = 0; i < n; i++)
    {
        for(j = 0; j < m; j++)
        {
            need[i][j] = max[i][j] - alloc[i][j];
        }
    }

    // Display Need Matrix
    printf("Need Matrix:\n");

    for(i = 0; i < n; i++)
    {
        printf("P%d ", i);

        for(j = 0; j < m; j++)
        {
            printf("%d ", need[i][j]);
        }

        printf("\n");
    }

    // Banker's Safety Algorithm
    while(count < n)
    {
        found = 0;

        for(i = 0; i < n; i++)
        {
            if(finish[i] == 0)
            {
                for(j = 0; j < m; j++)
                {
                    if(need[i][j] > avail[j])
                        break;
                }

                if(j == m)
                {
                    for(j = 0; j < m; j++)
                    {
                        avail[j] += alloc[i][j];
                    }

                    safeSeq[k++] = i;
                    finish[i] = 1;
                    count++;
                    found = 1;
                }
            }
        }

        if(found == 0)
        {
            break;
        }
    }

    // Display Result
    if(count == n)
    {
        printf("\nSystem is in SAFE state.\n");

        printf("Safe Sequence: ");

        for(i = 0; i < n; i++)
        {
            printf("P%d", safeSeq[i]);

            if(i != n - 1)
                printf(" -> ");
        }

        printf("\n");
    }
    else
    {
        printf("\nSystem is NOT in SAFE state.\n");
    }

    return 0;
}



5. How to run in Ubuntu Terminal
Follow these steps.

Step 1: Open Terminal

Press Ctrl + Alt + T.

Step 2: Create a C file
nano bankers.c


Step 3: Paste the above C program
- Right-click and paste the code.
- Press Ctrl + O and Enter to save.
- Press Ctrl + X to exit.
Step 4: Compile the program
gcc bankers.c -o bankers


Step 5: Execute the program
./bankers
